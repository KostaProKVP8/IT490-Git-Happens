#player (skaters) leaders 

curl -L -X GET "http://api-web.nhle.com/v1/skater-stats-leaders/current"

#team info (standings)
curl -L -X GET "http://api-web.nhle.com/v1/standings/now"

#all teams
curl -L -X GET "https://api.nhle.com/stats/rest/en/team"

#players on team
curl -L GET "http://api-web.nhle.com/v1/roster/{team}/current"

#score for a specified date

curl -L -X GET "api-web.nhle.com/v1/score/2026-MM-DD"
